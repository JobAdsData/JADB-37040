JADB 37040

Zooming in on the category 37040 (HISCO code).

Titles can be filtered through the ‘Filter titles’ box.

Titles can be selected by clicking or shift+clicking or cmd/ctrl+clicking on titles from the list.

The number in parenthesis is the total number of page hits for that title.

Inputs

database = FileAttachment("spring-37040.db.sqlite3").sqlite()

function k(d) {
  return Number(d3.utcFormat('%Y')(d)) +1
}
db = database.sql`SELECT year, std_title as label, counts, rel as relative, gender, journal_fk from data`

title_w_count = database.sql`select std_title as label, sum(counts) as count, journal_fk from data group by label, journal_fk`

journals = [ {id:1, name:'All of KB'}, {id:2, name:"Dagens Nyheter"} ]
genders = [ ... new Set(db.map((d) => d.gender))]
viewof journal = Inputs.select(journals,
  {label: html`Newspaper`,
    format: (t) => t.name,
    value: (t) => t.id,
    size : 2
  })

used_titles = title_w_count.filter((d) => (d.journal_fk == journal.id))

viewof titles = Inputs.select(reduced,
  {label: html`Titles`,
    value: (t) => t.label,
    format: (t) => `${t.label} (${t.count})`,
    multiple: true,
    size : 42,
    width: 350,
  })
titles_only = titles.map(d => d.label)

viewof title_target = Inputs.text({placeholder: "Filter titles"})

reduced = used_titles.filter((d) => (d.label.toLowerCase().includes(title_target)))
reduced_titles = reduced.map(d => d.label)
journal_filter = (d) => d.journal_fk == journal.id

title_filter = reduced.length === 0 ?  (d) => (true) : titles.length === 0 ? (d) => reduced_titles.includes(d.label) : (d) => titles_only.includes(d.label)


timeline_data = db.filter(journal_filter)
                    .filter(title_filter).map((d) => ({
                      year : new Date(d.year, 0, 1),
                      counts : d.counts,
                      title : d.title,
                      label : d.label,
                      relative : d.relative,
                    }))

Hits (pages)

Plot.plot({
    x: {
    label: "Year",
    tickformat: k,
  },
  y:{
    label: "Number of page hits",
    grid: true
  },
    // color: {legend: true},
    marks: [
      // Plot.ruleY([0, ylim]),

      Plot.lineY(timeline_data, {
        x: "year",
        y: (d) => d.counts,
        stroke:'label', curve: "catmull-rom"}, ),
      Plot.ruleX(timeline_data, Plot.pointerX({
        py: "counts",
        x: "year",
        stroke: 'red',
        format: k
        })),
    Plot.tip(timeline_data, Plot.pointer({
      x: "year",
      y: "counts",
      title: (d) => `${d.label}: ${d.value}\nYear: ${d.year}`
    }))
  ],
})
